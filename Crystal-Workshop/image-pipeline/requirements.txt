# requirements.txt — converter/image-pipeline
#
# Complete CPU-only image environment. Everything here installs from wheels on
# both Windows and the VPS (Ubuntu, no CUDA), so this one file makes rembg,
# GFPGAN and Real-ESRGAN available without installing NVIDIA libraries:
#
#   Windows:  py -3.11 -m venv .venv
#             .\.venv\Scripts\Activate.ps1
#             pip install -r requirements.txt
#
#   VPS:      uv venv --python 3.11 .venv
#             source .venv/bin/activate
#             pip install -r requirements.txt
#
# Model weights are downloaded once by code/download_models.py into the shared
# model cache. Releases and Python environments never contain duplicate weights.

# A flat, official wheel source only for the explicit +cpu requirements below.
# Unlike an extra package index, this cannot shadow ordinary PyPI dependencies.
--find-links https://download.pytorch.org/whl/cpu/torch_stable.html

# ── Background removal ────────────────────────────────────────────────────────
# The [cpu] extra selects onnxruntime over onnxruntime-gpu.
# 2.0.67 is the newest rembg release before its NumPy>=2.3 requirement. It
# already provides every six-model choice used by this app and shares the
# NumPy 1.26 ABI required by the stable Torch/BasicSR pair below.
rembg[cpu]==2.0.67
onnxruntime==1.28.0

# ── Imaging and CPU AI restoration/upscaling ─────────────────────────────────
pillow==12.3.0
PyMatting==1.1.15
scikit-image==0.26.0
torch==2.1.2+cpu
torchvision==0.16.2+cpu
basicsr==1.4.2
facexlib==0.3.0
gfpgan==1.3.8
realesrgan==0.3.0
opencv-python==4.10.0.84

# ── Numerics ──────────────────────────────────────────────────────────────────
# Torch 2.1's compiled NumPy bridge requires the final NumPy 1.x ABI. SciPy
# 1.17 supports 1.26.4, so all image engines share one stable ABI.
numpy==1.26.4
scipy==1.17.1

# ── Model download and caching ────────────────────────────────────────────────
pooch==1.9.0
requests==2.34.2
tqdm==4.70.0
